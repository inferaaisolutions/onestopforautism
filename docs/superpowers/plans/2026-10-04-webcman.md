# WebCMan Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build WebCMan — a serverless AI content manager that auto-updates all JSON content on any website daily via GitHub Actions, validated on the ASD Guide site.

**Architecture:** A `webcman/src/` directory of focused TypeScript modules wired together by an update engine. GitHub Actions runs `webcman/src/run.ts` on a cron schedule, which fetches from trusted sources, extracts plain-language content with Claude Haiku, writes JSON files, and commits changes back to the repo. A static HTML dashboard is generated after each run.

**Tech Stack:** TypeScript, tsx, Vitest, `@anthropic-ai/sdk` (claude-haiku-4-5-20251001), `fast-xml-parser`, GitHub Actions

```typescript
// webcman/src/schema-inferrer.ts
import fs from "fs";
import path from "path";
import type { SiteSchema, ContentSchema, FieldInfo } from "./types";

function detectType(value: unknown): FieldInfo["type"] {
  if (Array.isArray(value)) return "array";
  if (value === null) return "string";
  return typeof value as FieldInfo["type"];
}

function inferSchema(files: Record<string, unknown>[]): ContentSchema {
  if (files.length === 0) return {};
  const fieldCounts: Record<string, number> = {};
  const fieldTypes: Record<string, FieldInfo["type"]> = {};

  for (const file of files) {
    for (const [key, val] of Object.entries(file)) {
      fieldCounts[key] = (fieldCounts[key] ?? 0) + 1;
      fieldTypes[key] = detectType(val);
    }
  }

  const schema: ContentSchema = {};
  for (const [key, count] of Object.entries(fieldCounts)) {
    schema[key] = {
      type: fieldTypes[key],
      required: count / files.length >= 0.8,
    };
  }
  return schema;
}

export function inferSiteSchema(contentDir: string): SiteSchema {
  const siteSchema: SiteSchema = {};
  if (!fs.existsSync(contentDir)) return siteSchema;

  for (const subdir of fs.readdirSync(contentDir)) {
    const dirPath = path.join(contentDir, subdir);
    if (!fs.statSync(dirPath).isDirectory()) continue;

    const files: Record<string, unknown>[] = [];
    for (const file of fs.readdirSync(dirPath).filter(f => f.endsWith(".json"))) {
      try {
        files.push(JSON.parse(fs.readFileSync(path.join(dirPath, file), "utf-8")));
      } catch { /* skip malformed */ }
    }
    if (files.length > 0) siteSchema[subdir] = inferSchema(files);
  }
  return siteSchema;
}
```

```typescript
// webcman/src/trust-scorer.ts
import type { WebCManConfig } from "./types";

export function scoreDomain(domain: string, config: WebCManConfig): number {
  const d = domain.replace(/^www\./, "");
  const tiers = config.trustTiers;
  if (tiers.tier1.some(t => d.includes(t))) return 1;
  if (tiers.tier2.some(t => d.includes(t))) return 2;
  if (tiers.tier3.some(t => d.includes(t))) return 3;
  return 4;
}

export function scoreUrl(url: string, config: WebCManConfig): number {
  try {
    return scoreDomain(new URL(url).hostname, config);
  } catch {
    return 4;
  }
}
```

```typescript
// webcman/src/safety-filter.ts

export function isSafe(text: string, blockedPhrases: string[]): boolean {
  const lower = text.toLowerCase();
  return !blockedPhrases.some(phrase => lower.includes(phrase.toLowerCase()));
}

export function sanitize(text: string): string {
  return text.replace(/\s+/g, " ").trim();
}

export function isSafeObject(obj: Record<string, unknown>, blockedPhrases: string[]): boolean {
  const allText = JSON.stringify(obj).toLowerCase();
  return !blockedPhrases.some(phrase => allText.includes(phrase.toLowerCase()));
}
```

```typescript
// webcman/src/fetchers/rss.ts
import type { RawItem } from "../types";

function stripHtml(str: string): string {
  return str
    .replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/&amp;/g, "&")
    .replace(/&quot;/g, '"').replace(/&#39;/g, "'").replace(/&nbsp;/g, " ")
    .replace(/<[^>]+>/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}

function parseDate(raw: string | undefined): string {
  if (!raw) return new Date().toISOString().slice(0, 10);
  const d = new Date(raw);
  return isNaN(d.getTime()) ? new Date().toISOString().slice(0, 10) : d.toISOString().slice(0, 10);
}

export async function fetchRssItems(
  feedUrl: string,
  sourceDomain: string,
  trustScore: number
): Promise<RawItem[]> {
  try {
    const res = await fetch(feedUrl);
    if (!res.ok) return [];
    const xml = await res.text();
    const items: RawItem[] = [];
    const itemRegex = /<item>([\s\S]*?)<\/item>/g;
    let match: RegExpExecArray | null;
    while ((match = itemRegex.exec(xml))) {
      const block = match[1];
      const title = stripHtml(/<title>([\s\S]*?)<\/title>/.exec(block)?.[1] ?? "");
      const url = /<link>([\s\S]*?)<\/link>/.exec(block)?.[1]?.trim() ?? "";
      const content = stripHtml(/<description>([\s\S]*?)<\/description>/.exec(block)?.[1] ?? "") || title;
      const rawDate = /<pubDate>([\s\S]*?)<\/pubDate>/.exec(block)?.[1]?.trim();
      if (title && url) {
        items.push({ title, url, content, date: parseDate(rawDate), sourceDomain, trustScore });
      }
    }
    return items;
  } catch {
    return [];
  }
}
```

```typescript
// webcman/src/source-discovery.ts
import fs from "fs";
import path from "path";
import Anthropic from "@anthropic-ai/sdk";
import type { DiscoveredSource } from "./types";

const CACHE_TTL_DAYS = 30;

const DISCOVERY_PROMPT = (topic: string) => `You are a web researcher.
List the best publicly available RSS feeds and PubMed search URLs for the topic: "${topic}".

Rules:
- Only include feeds from well-known, trusted organizations (government health agencies, peer-reviewed journals, established non-profits)
- Include at least one PubMed RSS search URL
- Return 6-10 sources maximum
- Return ONLY a JSON array, no explanation

Format:
[
  { "feedUrl": "https://...", "domain": "example.com", "type": "rss" },
  { "feedUrl": "https://pubmed.ncbi.nlm.nih.gov/rss/search/?term=...", "domain": "pubmed.ncbi.nlm.nih.gov", "type": "pubmed" }
]`;

function isCacheFresh(cachePath: string): boolean {
  if (!fs.existsSync(cachePath)) return false;
  const stats = fs.statSync(cachePath);
  const ageMs = Date.now() - stats.mtimeMs;
  return ageMs < CACHE_TTL_DAYS * 24 * 60 * 60 * 1000;
}

export async function discoverSources(
  topic: string,
  apiKey: string,
  webcmanDir: string
): Promise<DiscoveredSource[]> {
  const cachePath = path.join(webcmanDir, "sources.json");

  if (isCacheFresh(cachePath)) {
    return JSON.parse(fs.readFileSync(cachePath, "utf-8")) as DiscoveredSource[];
  }

  const client = new Anthropic({ apiKey });
  const msg = await client.messages.create({
    model: "claude-haiku-4-5-20251001",
    max_tokens: 1024,
    messages: [{ role: "user", content: DISCOVERY_PROMPT(topic) }],
  });

  const text = msg.content.find(b => b.type === "text")?.text ?? "[]";
  let sources: DiscoveredSource[] = [];
  try {
```

```typescript
// webcman/src/extractor.ts
import Anthropic from "@anthropic-ai/sdk";
import type { RawItem, ContentSchema } from "./types";

const SYSTEM_PROMPT = `You are a content editor for a public information website.
Extract structured content from source articles.

STRICT RULES — any violation causes your output to be discarded:
- Write in plain language any general reader can understand
- No jargon; if a technical term is needed, define it briefly in parentheses
- NEVER write medical advice, legal advice, or financial advice
- NEVER write: "consult your doctor", "seek medical attention", "medical advice", "clinical recommendation"
- ALWAYS use informational phrasing: "research suggests", "a study found", "evidence indicates"
- ALWAYS include the source URL in the sources array
- If the article contradicts commonly known information on the same topic, add "conflictFlag": true
- Set sourceType to "scraped"
- Output ONLY valid JSON — no markdown fences, no explanation`;

function buildPrompt(item: RawItem, contentType: string, schema: ContentSchema): string {
  const schemaDescription = Object.entries(schema)
    .map(([k, v]) => `  "${k}": ${v.type}${v.required ? " (required)" : " (optional)"}`)
    .join("\n");

  return `Content type: ${contentType}

Source article:
Title: ${item.title}
URL: ${item.url}
Date: ${item.date}
Content: ${item.content}

Extract this into JSON with these fields:
{
${schemaDescription}
}

Source URL to include: ${item.url}`;
}

function parseJSON(text: string): Record<string, unknown> | null {
  try {
    const cleaned = text.trim().replace(/^```(?:json)?\s*/i, "").replace(/\s*```$/, "");
    return JSON.parse(cleaned) as Record<string, unknown>;
  } catch {
    return null;
  }
}

export async function extractItem(
  item: RawItem,
  contentType: string,
  schema: ContentSchema,
  apiKey: string
): Promise<Record<string, unknown> | null> {
  const client = new Anthropic({ apiKey });
  const prompt = buildPrompt(item, contentType, schema);

  for (let attempt = 0; attempt < 2; attempt++) {
    try {
      const msg = await client.messages.create({
        model: "claude-haiku-4-5-20251001",
        max_tokens: 1024,
        system: SYSTEM_PROMPT,
        messages: [{ role: "user", content: prompt }],
      });
      const text = msg.content.find(b => b.type === "text")?.text ?? "";
      const parsed = parseJSON(text);
      if (parsed) {
        parsed["sourceType"] = "scraped";
        return parsed;
      }
    } catch { /* retry */ }
  }
  return null;
}
```

```typescript
// webcman/src/history.ts
import fs from "fs";
import path from "path";

function historyDir(relPath: string, webcmanDir: string): string {
  const slug = relPath.replace(/[/\\]/g, "__").replace(/\.json$/, "");
  return path.join(webcmanDir, "history", slug);
}

export function saveVersion(
  relPath: string,
  content: unknown,
  webcmanDir: string,
  rollbackDepth: number
): void {
  const dir = historyDir(relPath, webcmanDir);
  fs.mkdirSync(dir, { recursive: true });
  const timestamp = new Date().toISOString().replace(/[:.]/g, "-");
  fs.writeFileSync(path.join(dir, `${timestamp}.json`), JSON.stringify(content, null, 2));

  // Prune oldest versions beyond rollbackDepth
  const versions = fs.readdirSync(dir).sort();
  while (versions.length > rollbackDepth) {
    fs.unlinkSync(path.join(dir, versions.shift()!));
  }
}

export function listVersions(relPath: string, webcmanDir: string): string[] {
  const dir = historyDir(relPath, webcmanDir);
  if (!fs.existsSync(dir)) return [];
  return fs.readdirSync(dir).sort().reverse(); // newest first
}

export function restoreVersion(
  relPath: string,
  timestamp: string,
  webcmanDir: string,
  contentDir: string
): boolean {
  const dir = historyDir(relPath, webcmanDir);
  const versionFile = fs.readdirSync(dir).find(f => f.startsWith(timestamp));
  if (!versionFile) return false;
  const content = fs.readFileSync(path.join(dir, versionFile), "utf-8");
  const target = path.join(contentDir, relPath);
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, content);
  return true;
}
```

```typescript
// webcman/src/update-engine.ts
import fs from "fs";
import path from "path";
import { discoverSources } from "./source-discovery";
import { fetchRssItems } from "./fetchers/rss";
import { fetchPubMedItems } from "./fetchers/pubmed";
import { scoreUrl } from "./trust-scorer";
import { isSafeObject, sanitize } from "./safety-filter";
import { extractItem } from "./extractor";
import { saveVersion } from "./history";
import type { WebCManConfig, SiteSchema, RawItem, RunStats } from "./types";

interface EngineOptions {
  config: WebCManConfig;
  apiKey: string;
  webcmanDir: string;
  contentTypes: string[];
  siteSchema: SiteSchema;
}

function getExistingUrls(contentTypeDir: string): Set<string> {
  const urls = new Set<string>();
  if (!fs.existsSync(contentTypeDir)) return urls;
  for (const file of fs.readdirSync(contentTypeDir).filter(f => f.endsWith(".json"))) {
    try {
      const item = JSON.parse(fs.readFileSync(path.join(contentTypeDir, file), "utf-8")) as Record<string, unknown>;
      const sources = item["sources"] as Array<{ url: string }> | undefined;
      sources?.forEach(s => urls.add(s.url));
    } catch { /* skip */ }
  }
  return urls;
}

function makeSlug(title: string): string {
  return title.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)/g, "").slice(0, 60);
}

export async function runUpdateEngine(opts: EngineOptions): Promise<RunStats> {
  const stats: RunStats = { added: [], updated: [], flagged: [], blocked: [], failed: [] };
  const { config, apiKey, webcmanDir, contentTypes, siteSchema } = opts;

  const sources = await discoverSources(config.topic, apiKey, webcmanDir);

  for (const contentType of contentTypes) {
    const schema = siteSchema[contentType];
    if (!schema) continue;

    const contentTypeDir = path.join(config.contentDir, contentType);
    const existingUrls = getExistingUrls(contentTypeDir);

    // Gather raw items from all sources
    const allItems: RawItem[] = [];
    for (const source of sources) {
      if (source.type === "pubmed") {
        const items = await fetchPubMedItems(config.topic);
        allItems.push(...items);
      } else {
        const score = scoreUrl(source.feedUrl, config);
        const items = await fetchRssItems(source.feedUrl, source.domain, score);
        allItems.push(...items);
      }
    }

    // Deduplicate against existing
    const seen = new Set(existingUrls);
    const fresh = allItems.filter(item => {
      if (seen.has(item.url)) return false;
      seen.add(item.url);
      return true;
    });

    // Extract and write
    for (const rawItem of fresh) {
      try {
        const extracted = await extractItem(rawItem, contentType, schema, apiKey);
        if (!extracted) {
          stats.failed.push(`Extraction failed: ${rawItem.title}`);
          continue;
        }

        if (!isSafeObject(extracted, config.blockedPhrases)) {
          stats.blocked.push(`Blocked: ${rawItem.title}`);
          continue;
        }

        const slug = (extracted["slug"] as string) || makeSlug(rawItem.title);
        const relPath = `${contentType}/${slug}.json`;
        const filePath = path.join(config.contentDir, relPath);

        // Save history if file exists
        if (fs.existsSync(filePath)) {
          const existing = JSON.parse(fs.readFileSync(filePath, "utf-8"));
          saveVersion(relPath, existing, webcmanDir, config.rollbackDepth);
          stats.updated.push(relPath);
        } else {
          stats.added.push(relPath);
        }

        if (extracted["conflictFlag"]) stats.flagged.push(relPath);

        fs.mkdirSync(contentTypeDir, { recursive: true });
        fs.writeFileSync(filePath, JSON.stringify({ ...extracted, slug, sourceType: "scraped" }, null, 2));
      } catch (err) {
        stats.failed.push(`Error for "${rawItem.title}": ${String(err)}`);
      }
    }
  }

  return stats;
}
```

```typescript
const CONFIG_PATH = path.join(process.cwd(), "webcman.config.json");

async function main(): Promise<void> {
  const apiKey = process.env.ANTHROPIC_API_KEY ?? "";
  if (!apiKey) {
    console.error("[WebCMan] ANTHROPIC_API_KEY is not set. Exiting.");
    process.exit(1);
  }

  console.log("[WebCMan] Loading config...");
  const config = loadConfig(CONFIG_PATH);

  console.log("[WebCMan] Inferring content schemas...");
  const siteSchema = inferSiteSchema(config.contentDir);
  const contentTypes = Object.keys(siteSchema);
  console.log(`[WebCMan] Found content types: ${contentTypes.join(", ")}`);

  console.log("[WebCMan] Running update engine...");
  const stats = await runUpdateEngine({
    config,
    apiKey,
    webcmanDir: WEBCMAN_DIR,
    contentTypes,
    siteSchema,
  });

  const logger = createRunLogger(WEBCMAN_DIR);
  stats.added.forEach(f => logger.logAdded(f));
  stats.updated.forEach(f => logger.logUpdated(f));
  stats.flagged.forEach(f => logger.logFlagged(f));
  stats.blocked.forEach(b => logger.logBlocked(b));
  stats.failed.forEach(f => logger.logFailed(f));
  logger.save();

  console.log(`[WebCMan] Done. +${stats.added.length} added, ~${stats.updated.length} updated, ${stats.flagged.length} flagged, ${stats.blocked.length} blocked, ${stats.failed.length} failed.`);

  // Generate dashboard
  const logsDir = path.join(WEBCMAN_DIR, "logs");
  const logEntries = fs.existsSync(logsDir)
    ? fs.readdirSync(logsDir).sort().reverse().slice(0, 30).map(f => {
        try { return JSON.parse(fs.readFileSync(path.join(logsDir, f), "utf-8")); } catch { return null; }
      }).filter(Boolean)
    : [];

  const dashboardHtml = generateDashboard(logEntries as Array<{ date: string; stats: typeof stats }>);
  fs.mkdirSync(path.join(process.cwd(), "public"), { recursive: true });
  fs.writeFileSync(path.join(process.cwd(), "public", "webcman.html"), dashboardHtml);
  console.log("[WebCMan] Dashboard written to public/webcman.html");

  // Git commit if running in CI
  if (process.env.CI) {
    const added = stats.added.length;
    const updated = stats.updated.length;
    const flagged = stats.flagged.length;
    const blocked = stats.blocked.length;
    if (added + updated > 0) {
      execSync("git config user.name 'webcman-bot'");
      execSync("git config user.email 'webcman@users.noreply.github.com'");
      execSync("git add content/ public/webcman.html .webcman/");
      execSync(`git commit -m "webcman: add ${added}, update ${updated}, flag ${flagged} conflicts, block ${blocked}"`);
      execSync("git push");
      console.log("[WebCMan] Committed and pushed changes.");
    } else {
      console.log("[WebCMan] No changes to commit.");
    }
  }
}

main().catch(err => {
  console.error("[WebCMan] Fatal error:", err);
  process.exit(1);
});
```
